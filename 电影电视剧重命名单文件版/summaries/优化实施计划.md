### 缓存优化实施

```python
# 1. 修改缓存TTL
class MediaRenamer:
    def __init__(self, settings):
        self.cache_ttl = settings.get('cache_ttl', 3600)  # 1小时
    
    def should_refresh_cache(self, folder_path: str) -> bool:
        # 检查文件夹修改时间
        if folder_path not in self.folder_cache:
            return True
        
        cache_time = self.folder_cache[folder_path]['timestamp']
        folder_mtime = os.path.getmtime(folder_path)
        
        return (time.time() - cache_time) > self.cache_ttl or folder_mtime > cache_time
```

### 扫描性能优化实施
```python
# 2. 并行扫描实现
from concurrent.futures import ThreadPoolExecutor
import os

class ScanWorker(QThread):
    def __init__(self):
        super().__init__()
        self.max_workers = min(4, os.cpu_count())
    
    def scan_folder_parallel(self, folder_path: str):
        files = self.get_all_files(folder_path)
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = []
            for file_path in files:
                future = executor.submit(self.analyze_file, file_path)
                futures.append(future)
            
            results = []
            for future in futures:
                try:
                    result = future.result(timeout=30)
                    if result:
                        results.append(result)
                except Exception as e:
                    logging.error(f"分析文件失败: {e}")
        
        return results
```
