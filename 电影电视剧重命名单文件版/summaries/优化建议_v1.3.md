### 11. 性能监控增强
**问题**: 需要更详细的性能监控
**建议**:
```python
# 在 main_window.py 中增强性能监控
class PerformanceMonitor:
    def __init__(self):
        self.metrics = {}
        self.start_time = time.time()
    
    def start_operation(self, operation: str):
        self.metrics[operation] = {
            'start_time': time.time(),
            'memory_start': psutil.Process().memory_info().rss
        }
    
    def end_operation(self, operation: str):
        if operation in self.metrics:
            end_time = time.time()
            memory_end = psutil.Process().memory_info().rss
            
            self.metrics[operation].update({
                'duration': end_time - self.metrics[operation]['start_time'],
                'memory_used': memory_end - self.metrics[operation]['memory_start']
            })
    
    def generate_report(self) -> Dict:
        # 生成性能报告
        return {
            'total_time': time.time() - self.start_time,
            'operations': self.metrics,
            'memory_peak': max([m.get('memory_used', 0) for m in self.metrics.values()])
        }
```
